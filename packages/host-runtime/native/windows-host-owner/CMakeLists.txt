cmake_minimum_required(VERSION 3.24)
project(octant_windows_host_owner LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Windows MSVC is required")
endif()
if(NOT OCTANT_NODE_HEADERS OR NOT OCTANT_NODE_IMPORT_LIB)
  message(FATAL_ERROR "Supply the existing Node-API SDK headers and node.lib")
endif()
add_library(octant-windows-host-owner SHARED owner.cpp)
set_target_properties(octant-windows-host-owner PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(octant-windows-host-owner PRIVATE "${OCTANT_NODE_HEADERS}")
target_compile_features(octant-windows-host-owner PRIVATE cxx_std_17)
target_compile_definitions(octant-windows-host-owner PRIVATE NAPI_VERSION=8 UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set_property(TARGET octant-windows-host-owner PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(octant-windows-host-owner PRIVATE /W4 /WX /EHsc /utf-8)
target_link_options(octant-windows-host-owner PRIVATE /DELAYLOAD:node.exe)
target_link_libraries(octant-windows-host-owner PRIVATE "${OCTANT_NODE_IMPORT_LIB}" delayimp advapi32 bcrypt shell32 ole32)
