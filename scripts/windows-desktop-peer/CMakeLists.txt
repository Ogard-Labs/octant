cmake_minimum_required(VERSION 3.24)
project(octant_windows_peer_qa LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Windows MSVC is required")
endif()
add_executable(octant-windows-peer-launcher launch.cpp)
add_executable(octant-windows-peer-appcontainer appcontainer.cpp)
add_executable(octant-windows-desktop-artifact-qa artifact.cpp)
foreach(target octant-windows-peer-launcher octant-windows-peer-appcontainer octant-windows-desktop-artifact-qa)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(${target} PRIVATE advapi32)
endforeach()
target_link_libraries(octant-windows-peer-appcontainer PRIVATE userenv bcrypt ole32)
target_link_libraries(octant-windows-desktop-artifact-qa PRIVATE shell32 ole32)
