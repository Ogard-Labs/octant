cmake_minimum_required(VERSION 3.24)
project(octant_desktop_peer_windows LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Native Windows MSVC is required")
endif()
if(NOT OCTANT_NODE_HEADERS OR NOT OCTANT_NODE_IMPORT_LIB)
  message(FATAL_ERROR "Supply the existing Node-API SDK headers and node.lib")
endif()
add_library(octant-desktop-peer-windows SHARED ../../src/native/desktopPeerWin32.cc)
set_target_properties(octant-desktop-peer-windows PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(octant-desktop-peer-windows PRIVATE "${OCTANT_NODE_HEADERS}")
target_compile_features(octant-desktop-peer-windows PRIVATE cxx_std_17)
target_compile_definitions(octant-desktop-peer-windows PRIVATE NAPI_VERSION=8 UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
set_property(TARGET octant-desktop-peer-windows PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(octant-desktop-peer-windows PRIVATE /W4 /WX /EHsc /utf-8)
target_link_options(octant-desktop-peer-windows PRIVATE /DELAYLOAD:node.exe)
target_link_libraries(octant-desktop-peer-windows PRIVATE "${OCTANT_NODE_IMPORT_LIB}" delayimp advapi32 bcrypt)

# Disposable hosted QA only; never stage these executables in app resources.
add_subdirectory(../../../../scripts/windows-desktop-peer "${CMAKE_CURRENT_BINARY_DIR}/qa")
